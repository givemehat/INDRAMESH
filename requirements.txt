# ---- IndraMesh runtime dependencies ----

# ---- TEST-ONLY dependencies ----
# These are imported by tests/, never by engine/. They are declared HERE rather than in a
# separate requirements-dev.txt because the CI workflow installs only requirements.txt plus
# `pytest`, so an undeclared test dependency produces a collection error on a fresh clone and
# in CI -- which is exactly what happened with hypothesis before this entry was added.
pytest>=8.0

# Property-based testing. tests/test_properties.py and tests/test_differential.py generate
# inputs with `derandomize=True` so every failure is reproducible from the test files alone.
hypothesis>=6.100

# ASGI test client. tests/test_server_api.py drives the FastAPI console through
# `fastapi.testclient.TestClient`, which imports httpx at collection time; an undeclared
# httpx fails the whole suite at collection on a fresh clone, exactly as hypothesis did.
httpx>=0.27
